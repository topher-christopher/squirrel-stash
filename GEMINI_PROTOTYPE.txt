function ChibiSquirrel({

  y = 0,

  carryingAcorn = false,
  golden = false,
  chefHat = false,
  scholarHat = false,
  scale = 1,
  bobOffset = 0,
}) {
  const furMain = golden ? '#F59E0B' : '#C26D38';
  const furBelly = golden ? '#FEF3C7' : '#FDE68A';
  const tailFill = golden ? '#D97706' : '#9A4E22';

  return (
    <g transform={`translate(${x}, ${y + bobOffset}) scale(${dir * scale}, ${scale})`}>
      {/* Soft Ground Shadow */}
      <ellipse cx="0" cy="14" rx="11" ry="3.5" fill="rgba(28, 17, 10, 0.28)" />

      {/* Plush Curled Tail */}
      <path
        d="M -5 10 C -19 10, -22 -7, -11 -11 C -5 -13, -3 -5, -7 -1 C -10 2, -6 6, -3 7 Z"
        fill={tailFill}
        stroke="#451A03"
        strokeWidth="1.2"
      />

      {/* Chibi Round Body */}
      <ellipse cx="0" cy="5" rx="8.5" ry="9" fill={furMain} stroke="#451A03" strokeWidth="1.2" />
      <ellipse cx="2.5" cy="6" rx="5" ry="6.2" fill={furBelly} />

      {/* Cozy Knitted Scarf (Appears during Frost) */}
      {scarf && (
        <g>
          <rect x="-7" y="-2.5" width="15" height="4.2" rx="2" fill="#EF4444" stroke="#7F1D1D" strokeWidth="0.9" />
          <path d="M -5 1 L -9 8 L -5 9 L -2 2 Z" fill="#DC2626" />
        </g>
      )}

      {/* Big Chibi Head */}
      <circle cx="1.5" cy="-7" r="8.2" fill={furMain} stroke="#451A03" strokeWidth="1.2" />

      {/* Ears */}
      <ellipse cx="-3.5" cy="-14.5" rx="2.4" ry="4" fill={furMain} stroke="#451A03" strokeWidth="1.1" />
      <ellipse cx="4.5" cy="-14.5" rx="2.4" ry="4" fill={furMain} stroke="#451A03" strokeWidth="1.1" />

      {/* Optional Cosmetic Hats */}
      {chefHat && (
        <g transform="translate(1, -16)">
          <rect x="-5" y="-3" width="10" height="4" rx="1.5" fill="#FFFFFF" stroke="#78350F" strokeWidth="0.8" />
          <circle cx="-3" cy="-5" r="3.2" fill="#FFFFFF" />
          <circle cx="3" cy="-5" r="3.2" fill="#FFFFFF" />
          <circle cx="0" cy="-6.5" r="3.8" fill="#FFFFFF" />
        </g>
      )}
      {scholarHat && !chefHat && (
        <g transform="translate(1, -15)">
          <polygon points="-8,-2 0,-6 8,-2 0,2" fill="#1E293B" stroke="#FBBF24" strokeWidth="0.8" />
          <line x1="6" y1="-2" x2="7.5" y2="3" stroke="#FBBF24" strokeWidth="1.1" />
        </g>
      )}

      {/* Eyes & Rosy Cheeks */}
      {pose === 'sleep' ? (
        <g>
          <path d="M 1 -7.5 Q 3.2 -5.8 5.2 -7.5" fill="none" stroke="#451A03" strokeWidth="1.3" strokeLinecap="round" />
          <text x="8" y="-13" fontSize="7.5" fontWeight="800" fill="#FEF3C7" opacity="0.85">

      ) : (

          <circle cx="5.3" cy="-8.3" r="0.6" fill="#FFFFFF" />

      {/* Reading / Researcher Glasses */}
      {(pose === 'researcher' || pose === 'focus') && (

          <circle cx="4.8" cy="-7.6" r="2.8" fill="rgba(224,242,254,0.35)" stroke="#FBBF24" strokeWidth="1.1" />
          <line x1="-1" y1="-8" x2="2" y2="-7.6" stroke="#FBBF24" strokeWidth="1" />

      {/* Pose-Specific Props */}
      {pose === 'tea' && (

        </g>
      )}

      {pose === 'lantern' && (
        <g transform="translate(10, 2)">
          <circle cx="0" cy="3" r="7" fill="#FEF08A" opacity="0.35" />
          <rect x="-2.5" y="0" width="5" height="6" rx="1.2" fill="#FACC15" stroke="#78350F" strokeWidth="0.9" />

          <path d="M -4.4 -1.8 Q 0 -4.8 4.4 -1.8 Z" fill="#78350F" />
        </g>
      )}
    </g>
  );

        audioCtxRef.current = null;
      }
      setAmbientAudioOn(false);
      return;
    }
    try {
      const AudioContextClass = window.AudioContext || window.webkitAudioContext;
      const ctx = new AudioContextClass();
      const bufferSize = ctx.sampleRate * 3;
      const buffer = ctx.createBuffer(1, bufferSize, ctx.sampleRate);
      const data = buffer.getChannelData(0);
      for (let i = 0; i < bufferSize; i++) {
        const white = Math.random() * 2 - 1;
        const crackle = Math.random() > 0.996 ? (Math.random() * 2 - 1) * 0.65 : 0;
        data[i] = white * 0.035 + crackle;

      const noise = ctx.createBufferSource();
      noise.buffer = buffer;
      noise.loop = true;

      const filter = ctx.createBiquadFilter();
      filter.type = 'lowpass';
      filter.frequency.value = 520;

      const gain = ctx.createGain();
      gain.gain.value = 0.25;

      noise.connect(filter);
      filter.connect(gain);
      gain.connect(ctx.destination);
      noise.start();

      audioCtxRef.current = ctx;
      setAmbientAudioOn(true);
    } catch {
      setAmbientAudioOn(false);

                    >
                      + Deposit $500 to Amber Reserve
                    </button>
                  </div>
                )}

                {selectedLowerKey === 'vault' && (
                  <div className="space-y-3">
                    <div className="p-3 rounded-2xl bg-stone-900 border border-amber-500/25">
                      <div className="text-2xl font-black text-slate-200">${vaultBalance.toLocaleString()}</div>
                      <div className="text-xs text-slate-300 mt-1">
                        Tracked separately from daily cash: Graded Pokémon Slabs ($3,450) + Silver Bullion ($1,650)
                      </div>
                    </div>
                    <button
                      onClick={() => setVaultBalance((v) => v + 250)}
                      className="w-full py-2.5 rounded-xl bg-slate-700 hover:bg-slate-600 font-bold text-xs text-white cursor-pointer"
                    >
                      + Log New Slab / Bullion (+$250)
                    </button>
                  </div>
                )}

                {selectedLowerKey === 'retirement' && (
                  <div className="space-y-3">
                    <div className="p-3 rounded-2xl bg-stone-900 border border-amber-500/25">
                      <div className="text-2xl font-black text-yellow-300">${retirementBalance.toLocaleString()}</div>
                      <div className="text-xs text-amber-100/80 mt-1">
                        Roth IRA + 401(k) locked deep in the ancient roots for long-term compounding.
                      </div>
                    </div>
                    <button
                      onClick={() => setRetirementBalance((r) => r + 500)}
                      className="w-full py-2.5 rounded-xl bg-amber-700 hover:bg-amber-600 font-bold text-xs text-white cursor-pointer"
                    >
                      + Log Retirement Contribution (+$500)
                    </button>
                  </div>
                )}
              </div>
            )}
          </div>
        </div>
      )}
    </div>
  );
